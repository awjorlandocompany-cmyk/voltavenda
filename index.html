async function checkSubscription(){ const content = document.getElementById("appContent"); const required = document.getElementById("subscriptionRequired"); const paymentMessage = document.getElementById("paymentMessage"); content.classList.add("hidden"); required.classList.add("hidden"); const { data, error } = await db .from("subscriptions") .select("status,current_period_end") .eq("user_id", currentUser.id) .maybeSingle(); if(error){ console.error("Erro ao verificar assinatura:", error); required.classList.remove("hidden"); required.querySelector("p").textContent = "Não foi possível verificar sua assinatura. Tente novamente."; paymentMessage.textContent = "Erro ao consultar assinatura."; return false; } if(!data){ required.classList.remove("hidden"); paymentMessage.textContent = "Você ainda não possui uma assinatura."; return false; } const validStatus = ["active", "trialing"].includes(data.status); const validPeriod = !data.current_period_end || new Date(data.current_period_end).getTime() > Date.now(); if(!validStatus || !validPeriod){ required.classList.remove("hidden"); paymentMessage.textContent = "Sua assinatura não está ativa. Regularize para acessar o sistema."; return false; } required.classList.add("hidden"); content.classList.remove("hidden"); paymentMessage.textContent = "Assinatura ativa. Acesso liberado."; await loadQuotes(); return true; } async function startApp(user){ currentUser = user; document.getElementById("landing").classList.add("hidden"); document.getElementById("auth").classList.add("hidden"); document.getElementById("app").classList.remove("hidden"); document.getElementById("welcome").textContent = "Oficina: " + (user.user_metadata.business_name || user.email); await checkSubscription(); }
